# ============================================================================
# Production dependencies. Target runtime: Python 3.12 (see Dockerfile).
#
# The MVP was pinned to Python 3.9, which reached end of life in October 2025
# and capped FastAPI, Starlette and python-multipart below their patched
# releases. Containerising the runtime is what unblocks those upgrades, and
# the versions below are the patched ones.
#
# Local development on a machine that still only has Python 3.9 should use
# requirements-py39.txt, which is the set the committed test results were
# produced with.
# ============================================================================

pydantic==2.10.4
email-validator==2.2.0

# --- database ----------------------------------------------------------------
sqlalchemy==2.0.36
alembic==1.14.0
psycopg[binary,pool]==3.2.3  # Postgres driver; pool for the worker's long-lived conns

# against the project's published JWKS.

bcrypt==4.2.1

# --- storage / HTTP ----------------------------------------------------------
